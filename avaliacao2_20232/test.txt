public class AdapterAES256Request extends CriptographyAESRequest {

    private AES256 criptography;

    public AdapterAES256Request (AES256 criptography, Request request) {
        super(request);
        this.criptography = criptography;
    }

    // A lógica de criptografia ficará aqui ao invés do decorator, retornando enfim a mensagem criptografada
    @Override
    public byte[] getContent() {
        
        byte[] content = super.getContent();

        criptography.init();

        content = criptography.doFinal(content);

        return content;
    }
}
