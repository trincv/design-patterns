public abstract class Decorator extends Calendario {

    protected Calendario calendario;

    public Decorator(Calendario calendario) { this.calendario = calendario; }

    @Override
    public void adicionarEvento(Evento evento) {
        super.adicionarEvento(evento);
    }

}


public class MunicipalEvento extends Decorator {

    public MunicipalEvento(Calendario calendario) { super(calendario); }

    @Override
    public Collection<Evento> from(Timestamp timestamp1, Timestamp timestamp2) {

        Collection<Evento> eventos = this.calendario.from(timestamp1,timestamp2);
        
        CalendarioOnline online = new CalendarioOnline(uri);                     // site de eventos municipais
        eventos.addAll(online.consultar(timestamp1, timestamp2));

        return eventos;                                                         // retorno de eventos do calendário e de um calendario municipal
    }

    @Override
    public Collection<Evento> day(Timestamp day) {

        Collection<Evento> eventos = this.calendario.day(day);
        
        CalendarioOnline online = new CalendarioOnline(uri);                     // site de eventos municipais
        eventos.addAll(online.consultar(day));

        return eventos;                                                         // retorno de eventos do calendário e de um calendario municipal
    }

    @Override
    public Collection<Evento> today() {

        Collection<Evento> eventos = this.calendario.today();
        
        CalendarioOnline online = new CalendarioOnline(uri);                     // site de eventos municipais
        eventos.addAll(online.consultar(Timestamp.today));                           // Algo assim

        return eventos;                                                         // retorno de eventos do calendário e de um calendario municipal
    }

}
